//! The Z/4 carry operator of the cofiber $C(h_j)$, as a cochain operator from the sphere's
//! differential: for $z$ at $(n, s)$, the value at the generators $g$ of $P_{s+2}$ in degree
//! $t_z + 2^j + 1$ of
//! $$ \Phi(z)(g) = \sum_{g \to y \to x} z(x) \sum [Sq^1 \in a'b'] \cdot [\tau(a'', b'') : m_0 \mapsto m_{top}] $$
//! over the coefficients $a$ of $dg$ at $y$ and $b$ of $dy$ at $x$, the coproduct terms of $a$ and
//! $b$, and $\tau$ the 2-part of $\sigma(a'')\sigma(b'')$ acting through the antipode. Written in
//! `cofiber_features`' format with feature name `CARRY` (and `CARRYrev` for $\sigma(b'')\sigma(a'')$).
//!
//! # Usage
//! Prompts as `secondary` (S_2, save dir, max n, max s); env `CARRY_MAX_N`, `CARRY_MAX_S`.

use std::sync::Arc;

use algebra::{
    Algebra,
    module::{FDModule, Module},
    pair_algebra::PairAlgebra,
};
use ext::{
    chain_complex::{ChainComplex, FreeChainComplex},
    ext_algebra::tensor_resolution::{Antipode, FullCoproduct},
    utils::{load_module_json, query_module},
};
use fp::vector::FpVector;
use sseq::coordinates::Bidegree;

fn main() -> anyhow::Result<()> {
    ext::utils::init_logging()?;
    let res = Arc::new(query_module(Some(algebra::AlgebraType::Milnor), false)?);
    let max_n: i32 = std::env::var("CARRY_MAX_N").map_or(58, |v| v.parse().unwrap());
    let max_s: i32 = std::env::var("CARRY_MAX_S").map_or(28, |v| v.parse().unwrap());
    let algebra = res.algebra();
    algebra.compute_basis(max_n + max_s + 20);
    let cop = FullCoproduct::new(Arc::clone(&algebra));
    let chi = Antipode::new(Arc::clone(&algebra));
    let sq1 = (0..algebra.dimension(1)).next().unwrap();
    println!("j,stem,Adams_filtration,index,feature,value");
    for (j, name) in [(0, "C2"), (1, "Ceta"), (2, "Cnu"), (3, "Csigma")] {
        let top = 1 << j;
        let m = FDModule::from_json(Arc::clone(&algebra), &load_module_json(name)?)?;
        m.compute_basis(top + 2);
        // does tau (degree top) carry m0 to the top cell, through chi?
        let carries = |deg: i32, e: &algebra::pair_algebra::MilnorPairElement| -> bool {
            let mut hit = false;
            for (ti, c) in e.twos().iter().enumerate() {
                if c == 0 {
                    continue;
                }
                let ch = chi.apply(deg, ti);
                let mut v = FpVector::new(algebra.prime(), m.dimension(top));
                for (o, oc) in ch.iter_nonzero() {
                    m.act_on_basis(v.as_slice_mut(), oc, deg, o, 0, 0);
                }
                hit ^= v.entry(0) != 0;
            }
            hit
        };
        for s in 0..=max_s {
            for n in 0..=max_n {
                let b0 = Bidegree::n_s(n, s);
                let tb = Bidegree::n_s(n + top - 1, s + 2);
                if !res.has_computed_bidegree(b0) || !res.has_computed_bidegree(tb) {
                    continue;
                }
                let dim = res.number_of_gens_in_bidegree(b0);
                let tdim = res.number_of_gens_in_bidegree(tb);
                if dim == 0 || tdim == 0 {
                    continue;
                }
                let tz = b0.t();
                let tg = tb.t();
                let p1 = res.module(s + 1);
                let p0 = res.module(s);
                let d2 = res.differential(s + 2);
                let d1 = res.differential(s + 1);
                let mut feat = [vec![vec![0u32; tdim]; dim], vec![vec![0u32; tdim]; dim]];
                for g in 0..tdim {
                    let dg = d2.output(tg, g);
                    for ty in (tz + 1)..tg {
                        for yi in 0..p1.number_of_gens_in_degree(ty) {
                            let aoff = p1.generator_offset(tg, ty, yi);
                            let ad = tg - ty;
                            for ai in 0..algebra.dimension(ad) {
                                if dg.entry(aoff + ai) == 0 {
                                    continue;
                                }
                                let dy = d1.output(ty, yi);
                                let bd = ty - tz;
                                for x in 0..dim {
                                    let boff = p0.generator_offset(ty, tz, x);
                                    for bi in 0..algebra.dimension(bd) {
                                        if dy.entry(boff + bi) == 0 {
                                            continue;
                                        }
                                        for &(al, ali, ar, ari) in cop.terms(ad, ai).iter() {
                                            for &(bl, bli, br, bri) in cop.terms(bd, bi).iter() {
                                                if ar + br != top || al + bl != 1 {
                                                    continue;
                                                }
                                                let mut left = FpVector::new(algebra.prime(), algebra.dimension(1));
                                                algebra.multiply_basis_elements(left.as_slice_mut(), 1, al, ali, bl, bli);
                                                if left.entry(sq1) == 0 {
                                                    continue;
                                                }
                                                for (v, (f, sec)) in [((ar, ari), (br, bri)), ((br, bri), (ar, ari))]
                                                    .into_iter()
                                                    .enumerate()
                                                {
                                                    let mut e = algebra.new_pair_element(top);
                                                    algebra.sigma_multiply_basis(&mut e, 1, f.0, f.1, sec.0, sec.1);
                                                    if carries(top, &e) {
                                                        feat[v][x][g] ^= 1;
                                                    }
                                                }
                                            }
                                        }
                                    }
                                }
                            }
                        }
                    }
                }
                for (v, fname) in ["CARRY", "CARRYrev"].into_iter().enumerate() {
                    for (x, row) in feat[v].iter().enumerate() {
                        if row.iter().any(|&c| c != 0) {
                            println!("{j},{n},{s},{x},{fname},\"{row:?}\"");
                        }
                    }
                }
            }
        }
    }
    Ok(())
}
